    function parseBirthday(bday) {
      // Accept "MM-DD" or "YYYY-MM-DD"
      if (!bday || bday.trim() === "" || bday === "MM-DD") return null;
      if (/^\d{2}-\d{2}$/.test(bday)) {
        const [MM,DD] = bday.split('-');
        const now = new Date();
        let year = now.getFullYear();
        let target = new Date(`${year}-${MM}-${DD}T00:00:00`);
        // If already passed this year, pick next year

        return target;

        return new Date(bday + 'T00:00:00');

      return null;
